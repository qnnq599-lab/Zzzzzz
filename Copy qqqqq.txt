Inverse Discrete Fourier Transform (IDFT).
clc;
close all;
clear all;
xn = input('Enter the sequence x(n)'); % Get the sequence
N = length(xn); % Find the length of the sequence
xk = zeros(1,N); % Initialize array for DFT
ixk = zeros(1,N); % Initialize array for IDFT

% DFT of the sequence
for k = 0:N-1
    for n = 0:N-1
        xk(k+1) = xk(k+1) + (xn(n+1) * exp((i)*2*pi*k*n/N));
    end
end

% Plotting input sequence
t = 0:N-1;
subplot(2,2,1);
stem(t, xn);
ylabel('Amplitude');
xlabel('Time Index');
title('Input Sequence');

% Magnitude response
magnitude = abs(xk); % Find the magnitudes of individual components
t = 0:N-1;
subplot(2,2,2);
stem(t, magnitude);
ylabel('Amplitude');
title('Magnitude Response');
xlabel('K');

% Phase response
phase = angle(xk); % Find the phases of the system
t = 0:N-1;
subplot(2,2,3);
stem(t, phase);
ylabel('Phase');
xlabel('K');
title('Phase Response');

% IDFT of the sequence
for n = 0:N-1
    for k = 0:N-1
        ixk(n+1) = ixk(n+1) + (xk(k+1) * exp(I*2*pi*k*n/N));
    end
end
ixk = ixk ./ N;

% Plot IDFT output (reconstructed sequence)
t = 0:N-1;
subplot(2,2,4);
stem(t, ixk);
ylabel('Amplitude');
xlabel('Time index');
